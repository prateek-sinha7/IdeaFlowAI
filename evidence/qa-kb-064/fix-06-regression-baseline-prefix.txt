"""Baseline only: put back the PRE-FIX RunSandbox.list_tags body in-process (the working tree is untouched).
app is imported lazily in pytest_runtest_setup so the plugin cannot bind a live DB at collection."""


def pytest_runtest_setup(item):
    import app.agents.sandbox as sb

    def list_tags(self):
        if not self.has_git:
            return []
        fn = getattr(self._ws(), "list_tags", None)
        if fn is None:
            return []
        try:
            return fn()
        except Exception:  # noqa: BLE001
            return []

    sb.RunSandbox.list_tags = list_tags
