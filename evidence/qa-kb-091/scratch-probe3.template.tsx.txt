type Scn = { label: string; props: Record<string, unknown>; pin?: Record<string, unknown> };

async function runScenario(s: Scn) {
  mockGetWorkflow.mockReset();
  mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, ...(s.pin ?? { status: "completed", output: null }) } as any));
  render(<PreviewPanel workflowType="prototype" runFamily={tagless} {...(s.props as any)} />);
  if (s.pin) {
    await waitFor(() => { if (mockGetWorkflow.mock.calls.length === 0) throw new Error("pin fetch not issued"); });
    await screen.findAllByText(/read-only/i);
  } else {
    await new Promise((r) => setTimeout(r, 30));
  }
  await new Promise((r) => setTimeout(r, 30));
  const tabEls = screen.queryAllByRole("tab");
  const tabs = tabEls.map((t) => t.textContent?.trim());
  const landed = tabEls.find((t) => t.getAttribute("aria-selected") === "true")?.textContent?.trim() ?? "(none)";
  const txt0 = document.body.textContent || "";
  const f0 = flags(txt0);
  const share = !!screen.queryByLabelText("Share this run");
  const download = !!screen.queryByLabelText("Download the deliverable");
  const hdrFailed = txt0.includes("Run failed");
  const partial = txt0.includes("· partial");
  const proto = !!screen.queryByTestId("proto-preview");
  // Now force the Preview tab (if it is in the strip) and re-read what it renders.
  let afterPreview = "n/a (no Preview tab in strip)";
  const prev = tabEls.find((t) => t.textContent?.trim() === "Preview");
  if (prev) {
    fireEvent.click(prev);
    await new Promise((r) => setTimeout(r, 30));
    const f1 = flags(document.body.textContent || "");
    afterPreview = JSON.stringify({ ...f1, proto: !!screen.queryByTestId("proto-preview") });
  }
  const line = [
    "ROW " + s.label,
    "| tabs=" + JSON.stringify(tabs),
    "| landed=" + landed,
    "| landedCopy=" + JSON.stringify({ ...f0, proto }),
    "| PreviewTabClicked=" + afterPreview,
    "| header=" + JSON.stringify({ runFailed: hdrFailed, partialChip: partial, shareBtn: share, downloadBtn: download }),
  ].join(" ");
  fs.appendFileSync(OUT, line + "\n");
}

const tagged: any = { root_id: "root", members: [{ ...tagless.members[0], git_tag: "v1" }, { ...tagless.members[1], git_tag: "v2" }] };
async function menuFlow(label: string, props: Record<string, unknown>, pin: Record<string, unknown>) {
  mockGetRunSandboxRefs.mockReset().mockResolvedValue(["v1", "v2"]);
  mockGetWorkflow.mockReset();
  mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, ...pin } as any));
  const user = userEvent.setup();
  render(<PreviewPanel workflowType="prototype" runFamily={tagged} liveRunId="r1" {...(props as any)} />);
  await new Promise((r) => setTimeout(r, 50));
  const before = document.body.textContent || "";
  const beforeHdr = { runFailed: before.includes("Run failed"), share: !!screen.queryByLabelText("Share this run"), download: !!screen.queryByLabelText("Download the deliverable") };
  const chip = (await screen.findAllByRole("button")).find((b) => /v2/.test(b.textContent || ""));
  if (!chip) { fs.appendFileSync(OUT, "ROW " + label + " | no version chip button\n"); return; }
  await user.click(chip);
  await user.click(await screen.findByRole("option", { name: "v1" }));
  await waitFor(() => expect_(mockGetWorkflow.mock.calls.length > 0));
  await screen.findAllByText(/Viewing v1 \(read-only\)/);
  await new Promise((r) => setTimeout(r, 50));
  const txt = document.body.textContent || "";
  const tabs = screen.queryAllByRole("tab").map((x) => x.textContent?.trim());
  const after = { runFailed: txt.includes("Run failed"), partialChip: txt.includes("\u00b7 partial"), share: !!screen.queryByLabelText("Share this run"), download: !!screen.queryByLabelText("Download the deliverable"), tabs };
  fs.appendFileSync(OUT, "ROW " + label + " | BEFORE-pick(shell on screen)=" + JSON.stringify(beforeHdr) + " | AFTER-pick(v1 pinned, read-only)=" + JSON.stringify(after) + "\n");
}
