import { LaunchWizard } from "@/components/workflow/LaunchWizard";

const agents = (): AgentDef[] =>
  [...PROTO_AGENTS.map((id, i) => ({ id, name: id, role: "t", description: "t", pipeline_type: "prototype", order: i + 1, icon: "zap", estimated_duration: 60, has_skill: false })),
   ...PPT_AGENTS.map((id, i) => ({ id, name: id, role: "t", description: "t", pipeline_type: "ppt", order: i + 1, icon: "zap", estimated_duration: 60, has_skill: false }))] as AgentDef[];

const preloaded = () => ({
  agents: { agents: agents(), totalCount: agents().length, pipelines: {}, status: "succeeded" as const, error: null },
});

const settle = () => new Promise((r) => setTimeout(r, 80));

const TEMPLATES_OK = [{ id: "kanban", name: "Kanban", example_prompt: "Example brief" }];
const DECKS_OK = [
  { id: "pitch", name: "Pitch", design_system: { requires: true } },
  { id: "onepager", name: "One Pager", design_system: { requires: false } },
];

beforeEach(() => {
  sessionStorage.clear();
  vi.mocked(listPrototypeTemplates).mockReset().mockResolvedValue(TEMPLATES_OK as never);
  vi.mocked(listPPTTemplates).mockReset().mockResolvedValue(DECKS_OK as never);
  vi.mocked(listDesignSystems).mockReset().mockResolvedValue([{ id: "midnight", name: "Midnight" }] as never);
});

describe("QA-KB-081 design-system step: loading vs failed vs empty", () => {
  it("CONTROL: a loaded registry renders the picker and no skeleton", async () => {
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    expect(await screen.findByTestId("pick-ds")).toBeInTheDocument();
    expect(screen.getByTestId("ds-slot").querySelector(".animate-pulse")).toBeNull();
  });

  it("OBSERVE: while the fetch is still pending the skeleton is shown (this must survive any fix)", async () => {
    vi.mocked(listDesignSystems).mockReturnValue(new Promise(() => {}) as never);
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    await waitFor(() => expect(listDesignSystems).toHaveBeenCalled());
    expect(screen.getByTestId("ds-slot").querySelector(".animate-pulse")).not.toBeNull();
  });

  it("EXPECTED: registry REJECTS -> an error is shown and the skeleton is gone", async () => {
    vi.mocked(listDesignSystems).mockRejectedValue(new Error("500: boom"));
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    await waitFor(() => expect(listDesignSystems).toHaveBeenCalled());
    await settle();
    const slot = screen.getByTestId("ds-slot");
    // Report what is actually there before asserting, so the output is evidence.
    console.log("[reject arm] ds-slot html:", slot.innerHTML);
    expect(slot.querySelector(".animate-pulse")).toBeNull();
    expect(slot.textContent ?? "").toMatch(/couldn.t|could not|failed|error|unavailable|retry/i);
  });

  it("EXPECTED: registry resolves [] -> an empty message is shown and the skeleton is gone", async () => {
    vi.mocked(listDesignSystems).mockResolvedValue([] as never);
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    await waitFor(() => expect(listDesignSystems).toHaveBeenCalled());
    await settle();
    const slot = screen.getByTestId("ds-slot");
    console.log("[empty arm] ds-slot html:", slot.innerHTML);
    expect(slot.querySelector(".animate-pulse")).toBeNull();
    expect(slot.textContent ?? "").toMatch(/no design systems|none available|unavailable|empty/i);
  });

  it("OBSERVE: in the failed arm Continue stays disabled and the pill demands a design system", async () => {
    vi.mocked(listDesignSystems).mockRejectedValue(new Error("500: boom"));
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    await user.type(await screen.findByLabelText("Brief"), "Build a kanban board");
    await settle();
    expect(screen.getByRole("button", { name: "Continue" })).toBeDisabled();
    expect(screen.getByText("Pick a design system")).toBeInTheDocument();
  });

  it("OBSERVE: the custom-design-system escape hatch is not reachable in the empty arm (picker is not mounted)", async () => {
    vi.mocked(listDesignSystems).mockResolvedValue([] as never);
    renderWithProviders(<LaunchWizard initialMode="prototype" />, { preloadedState: preloaded() });
    await waitFor(() => expect(listDesignSystems).toHaveBeenCalled());
    await settle();
    expect(screen.queryByTestId("pick-ds")).toBeNull();
    expect(screen.queryByRole("button", { name: /upload custom/i })).toBeNull();
  });

  it("CONTROL: the REAL DesignSystemPicker given systems=[] DOES offer 'Upload custom' (so the picker could stay mounted)", async () => {
    const actual = await vi.importActual<typeof import("@/components/workflow/prototype/DesignSystemPicker")>(
      "@/components/workflow/prototype/DesignSystemPicker",
    );
    render(<actual.DesignSystemPicker systems={[]} selectedId={null} onSelect={vi.fn()} onSelectCustom={vi.fn()} />);
    expect(screen.getAllByRole("button", { name: /upload custom/i }).length).toBeGreaterThan(0);
    expect(screen.getByText("0 design systems")).toBeInTheDocument();
  });

  it("OBSERVE (deck, post FIX-695): a DS-requiring deck template mounts the DS step and it is the same skeleton when the registry fails", async () => {
    vi.mocked(listDesignSystems).mockRejectedValue(new Error("500: boom"));
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="ppt" />, { preloadedState: preloaded() });
    await waitFor(() => expect(listPPTTemplates).toHaveBeenCalled());
    await user.click(await screen.findByTestId("pick-deck"));
    await settle();
    expect(screen.getByTestId("stepper").getAttribute("data-steps")).toBe("template,design-system");
    expect(screen.getByTestId("ds-slot").querySelector(".animate-pulse")).not.toBeNull();
    expect(screen.getByText("Pick a design system")).toBeInTheDocument();
  });
});
