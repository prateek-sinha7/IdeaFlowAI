class _Cap(logging.Handler):
    def __init__(self):
        super().__init__()
        self.lines = []

    def emit(self, record):
        m = record.getMessage()
        if "resume_run(" in m or "_replay_clarify_run(" in m:
            if record.levelno >= logging.WARNING or "step offset" in m or "all steps complete" in m:
                self.lines.append(f"{record.levelname}:{m.split('): ', 1)[1]}")

def new_run(typ, status="generating"):
    _n["i"] += 1
    rid = f"q4-{typ}-{_n['i']}"
    session.add(WorkflowRun(id=rid, user_id=OWNER, owner_id=OWNER, workspace_id=WS,
                            status=status, type=typ, input="brief"))
    session.commit()
    return rid


async def drive(method, rid):
    engine = ExecutionEngine()
    seen = {"reached_drive": False}

    async def spy_drive(run_id, **kw):
        seen["reached_drive"] = True
        seen["roster"] = [a.id for a in kw["agents"]]
        seen["resume_from"] = kw.get("_resume_from")

    engine._drive_resumed_stream = spy_drive
    cap.lines.clear()
    await getattr(engine, method)(rid)
    return list(cap.lines), seen
