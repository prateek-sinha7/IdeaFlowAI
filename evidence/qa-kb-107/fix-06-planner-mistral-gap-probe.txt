import sys
from unittest.mock import MagicMock

import langchain_aws, langchain_mistralai, langchain_anthropic, langchain_ollama  # noqa: E401,E402
from app.core.config import settings  # noqa: E402

for mod, name in ((langchain_aws, "ChatBedrockConverse"), (langchain_mistralai, "ChatMistralAI"),
                  (langchain_anthropic, "ChatAnthropic"), (langchain_ollama, "ChatOllama")):
    m = MagicMock(name=name)
    setattr(mod, name, m)
    rec[name] = m
for k, v in (("ANTHROPIC_API_KEY", ""), ("BEDROCK_INFERENCE_PROFILE_ID", ""), ("BEDROCK_MODEL_ID", ""),
             ("MISTRAL_API_KEY", "m"), ("AWS_BEARER_TOKEN_BEDROCK", "")):
    setattr(settings, k, v)

from agents.planner.smart_planner import SmartPlanner  # noqa: E402


def called():
    out = [n for n, m in rec.items() if m.called]
    for m in rec.values():
        m.reset_mock()
    return out


build_model()
