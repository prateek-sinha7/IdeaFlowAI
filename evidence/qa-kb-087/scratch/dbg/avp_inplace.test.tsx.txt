import { describe, it, vi, beforeEach, afterEach } from "vitest";
import { render, cleanup } from "@testing-library/react";

const getRunArtifacts = vi.fn();

  getRunArtifacts: (...args: unknown[]) => getRunArtifacts(...args),

import { ArtifactVersionPicker } from "@/components/results/ArtifactVersionPicker";

beforeEach(() => {
  getRunArtifacts.mockReset().mockResolvedValue({ workflow_id: "r1", artifacts: [

  ] });
});
afterEach(() => { cleanup(); vi.clearAllMocks(); });

describe("in-place resolution (real component file, real alias)", () => {

    render(<React.StrictMode><ArtifactVersionPicker runId="r1" agentId="prototype-specify" onSelect={() => {}} /></React.StrictMode>);
    const samples: number[] = [];
    for (let i = 0; i < 6; i++) { await new Promise((r) => setTimeout(r, 25)); samples.push(getRunArtifacts.mock.calls.length); }
    console.log("INPLACE STRICT calls over time:", JSON.stringify(samples));
  });

    render(<ArtifactVersionPicker runId="r1" agentId="prototype-specify" onSelect={() => {}} />);
    const samples: number[] = [];

    console.log("INPLACE PLAIN calls over time:", JSON.stringify(samples));
