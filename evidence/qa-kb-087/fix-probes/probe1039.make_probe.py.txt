# QA-KB-087 5-fixer probe for ISS-1039 (LibraryPage.copyFeedback) and ISS-1040 (IntegrationsCard.copyFeedback).
# Builds scratch copies next to this script from git HEAD; never writes the repo.
#   head_marker_off_* : HEAD test with only it.fails -> it   (the "red for the wrong reason" the cards describe)
#   fixed_*           : HEAD test with the card's full fix     (harness repair + it.fails -> it)
#   *_real / *_mutant : against the real component / a component with the Copy-failed feedback removed
import pathlib, subprocess
REPO = "/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2"
def head(rel): return subprocess.run(["git", "-C", REPO, "show", "HEAD:" + rel], capture_output=True, text=True, check=True).stdout
def work(rel): return pathlib.Path(REPO + "/" + rel).read_text()
def sub(s, a, b, n=1):
    assert s.count(a) == n, (a, s.count(a)); return s.replace(a, b)

# ---- ISS-1039
lt = head("frontend/src/components/library/LibraryPage.copyFeedback.test.tsx")
l_off = sub(lt, "it.fails(", "it(", 2)
l_fix = sub(l_off, "  useParams: () => ({}),\n", "  useParams: vi.fn(() => ({})),\n")
lp = work("frontend/src/components/library/LibraryPage.tsx")
pathlib.Path("MutantLibraryPage.tsx").write_text(sub(lp, "  const { copied, failed, copy } = useClipboardCopy();\n",
    "  const { copied, copy } = useClipboardCopy();\n  const failed = false;\n", 2))
for name, src, comp in [("lib_head_marker_off_real", l_off, "@/components/library/LibraryPage"),
                        ("lib_fixed_real", l_fix, "@/components/library/LibraryPage"),
                        ("lib_fixed_mutant", l_fix, "./MutantLibraryPage")]:
    pathlib.Path(f"{name}.test.tsx").write_text(sub(src, 'import { LibraryPage } from "./LibraryPage";', f'import {{ LibraryPage }} from "{comp}";'))

# ---- ISS-1040
it_ = head("frontend/src/components/handoff/IntegrationsCard.copyFeedback.test.tsx")
i_off = sub(it_, "it.fails(", "it(")
i_fix = sub(i_off, '.closest("div")!;', ".parentElement!;")
ic = work("frontend/src/components/handoff/IntegrationsCard.tsx")
pathlib.Path("MutantIntegrationsCard.tsx").write_text(sub(ic,
    "    copied: keyCopied,\n    failed: keyFailed,\n    copy: copyKey,\n  } = useClipboardCopy();\n",
    "    copy: copyKey,\n  } = useClipboardCopy();\n  const keyCopied = false, keyFailed = false;\n"))
for name, src, comp in [("int_head_marker_off_real", i_off, "@/components/handoff/IntegrationsCard"),
                        ("int_fixed_real", i_fix, "@/components/handoff/IntegrationsCard"),
                        ("int_fixed_mutant", i_fix, "./MutantIntegrationsCard")]:
    pathlib.Path(f"{name}.test.tsx").write_text(sub(src, 'import { IntegrationsCard } from "./IntegrationsCard";', f'import {{ IntegrationsCard }} from "{comp}";'))
print("ok")
