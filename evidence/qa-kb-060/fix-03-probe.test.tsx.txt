// TEMPORARY probe for QA-KB-060 fix phase. Deleted after the run; output kept in evidence/qa-kb-060/.
import { describe, it, expect, vi, beforeEach } from "vitest";
import { renderWithProviders, screen } from "@/test/renderWithProviders";
import userEvent from "@testing-library/user-event";
import type { WorkflowType } from "@/types/index";
import type { ManifestStep } from "@/store/api/userWorkflows";

const mockSaveUserWorkflow = vi.fn();
vi.mock("@/lib/api", () => ({
  getToken: () => "test-token",
  getCapabilities: vi.fn().mockResolvedValue({ capabilities: [], model_catalog: [] }),
  saveUserWorkflow: (...a: unknown[]) => mockSaveUserWorkflow(...a),
  getWorkflowDetail: vi.fn().mockResolvedValue({ steps: [] }),
  getAgentPrompt: vi.fn().mockResolvedValue({ prompt_body: "", override: null, has_override: false }),
  saveAgentPromptOverride: vi.fn(),
  deleteAgentPromptOverride: vi.fn(),
}));

import { ComposerPage } from "./ComposerPage";
import { SkillsHooksProvider } from "@/context/SkillsHooksContext";

const PPT_STEPS: ManifestStep[] = [{ agent: "ppt-brief-analyst" }, { agent: "ppt-composer" }];
function ui(props: Record<string, unknown>) {
  return (
    <SkillsHooksProvider>
      <ComposerPage workflowType={"custom" as WorkflowType} onBack={() => {}} {...props} />
    </SkillsHooksProvider>
  );
}
const render = (props: Record<string, unknown>) =>
  renderWithProviders(ui(props), {
    preloadedState: { agents: { agents: [], totalCount: 0, pipelines: {}, status: "succeeded", error: null } },
  });
const saveBtn = () => screen.getByRole("button", { name: /Save workflow/i });

beforeEach(() => {
  vi.clearAllMocks();
  mockSaveUserWorkflow.mockResolvedValue({ id: "wf" });
});

describe("QA-KB-060 probe", () => {
  it("A. plain rerender (no remount): a late initialUserWorkflowId never reaches the id handleSave sends", async () => {
    // No expectedUserWorkflowId, so the new gate is out of the picture (the pre-fix shape).
    const { rerender } = render({ initialName: "renamed", initialManifestSteps: PPT_STEPS });
    rerender(ui({ initialUserWorkflowId: "wf-123", initialName: "renamed", initialManifestSteps: PPT_STEPS }));
    await userEvent.click(saveBtn());
    console.log("A. saveUserWorkflow 2nd arg =", JSON.stringify(mockSaveUserWorkflow.mock.calls[0]?.[1]));
    expect(mockSaveUserWorkflow).toHaveBeenCalledTimes(1);
    expect(mockSaveUserWorkflow.mock.calls[0][1]).toBeUndefined(); // create branch: POST a new row
  });

  it("B. keyed remount (the DashboardLayout key flip): the late seed releases Save and it targets the URL's row", async () => {
    const { rerender } = render({ expectedUserWorkflowId: "wf-123", initialName: "renamed" });
    expect(saveBtn()).toBeDisabled();
    rerender(ui({ key: "wf-123", expectedUserWorkflowId: "wf-123", initialUserWorkflowId: "wf-123", initialName: "renamed", initialManifestSteps: PPT_STEPS }));
    expect(saveBtn()).toBeEnabled();
    await userEvent.click(saveBtn());
    console.log("B. saveUserWorkflow 2nd arg =", JSON.stringify(mockSaveUserWorkflow.mock.calls[0]?.[1]));
    expect(mockSaveUserWorkflow.mock.calls[0][1]).toBe("wf-123"); // update branch: PATCH the edited row
  });

  it("C. /workflows/new shape (no expectedUserWorkflowId): Save stays live and creates", async () => {
    render({ initialName: "new flow", initialManifestSteps: PPT_STEPS });
    expect(saveBtn()).toBeEnabled();
    await userEvent.click(saveBtn());
    console.log("C. saveUserWorkflow 2nd arg =", JSON.stringify(mockSaveUserWorkflow.mock.calls[0]?.[1]));
    expect(mockSaveUserWorkflow.mock.calls[0][1]).toBeUndefined();
  });

  it("D. failed /edit seed: tooltip names the load failure", () => {
    render({ expectedUserWorkflowId: "wf-123", initialName: "renamed", initialManifestSteps: PPT_STEPS });
    console.log("D. disabled =", (saveBtn() as HTMLButtonElement).disabled, "title =", JSON.stringify(saveBtn().getAttribute("title")));
    expect(saveBtn()).toBeDisabled();
  });
});
