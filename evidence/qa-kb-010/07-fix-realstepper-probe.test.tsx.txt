import { LaunchWizard as Current } from "@/components/workflow/LaunchWizard";
import { LaunchWizard as Head } from "./LaunchWizard.HEAD";

const agents = (): AgentDef[] => [
  ...PROTO_AGENTS.map((id, i) => ({ id, name: id, role: "r", description: "d", pipeline_type: "prototype", order: i + 1, icon: "zap", estimated_duration: 60, has_skill: false })),
  ...PPT_AGENTS.map((id, i) => ({ id, name: id, role: "r", description: "d", pipeline_type: "ppt", order: i + 1, icon: "zap", estimated_duration: 60, has_skill: false })),
] as AgentDef[];
const state = () => ({ agents: { agents: agents(), totalCount: agents().length, pipelines: {}, status: "succeeded" as const, error: null } });
const drafts = () => Object.keys(sessionStorage).filter((k) => k.endsWith(".draft")).map((k) => sessionStorage.getItem(k) ?? "");

beforeEach(() => sessionStorage.clear());

for (const [label, Wizard] of [["current", Current], ["head", Head]] as const) {
  for (const mode of ["ppt", "ppt_v2"] as const) {
    describe(`${label} ${mode}`, () => {
      it("DS-requiring template: DS tab mounts the picker, Continue enables, draft carries the DS", async () => {
        const user = userEvent.setup();
        renderWithProviders(<Wizard initialMode={mode} />, { preloadedState: state() });
        await user.type(await screen.findByLabelText("Brief"), "Pitch deck");
        await user.click(await screen.findByTestId("gallery-pick-pitch"));
        expect(screen.getByRole("button", { name: "Continue" })).toBeDisabled();
        expect(screen.getByText("Pick a design system")).toBeInTheDocument();
        await user.click(screen.getByTestId("tab-design-system"));
        await user.click(await screen.findByTestId("pick-ds"));
        expect(screen.queryByText("Pick a design system")).toBeNull();
        const cont = screen.getByRole("button", { name: "Continue" });
        expect(cont).toBeEnabled();
        await user.click(cont);
        expect(drafts().some((d) => d.includes('"designSystemId":"midnight"'))).toBe(true);
      });

      it("non-DS template: only the Template tab, Continue enables with no DS", async () => {
        const user = userEvent.setup();
        renderWithProviders(<Wizard initialMode={mode} />, { preloadedState: state() });
        await user.type(await screen.findByLabelText("Brief"), "Pitch deck");
        await user.click(await screen.findByTestId("gallery-pick-onepager"));
        expect(screen.queryByTestId("tab-design-system")).toBeNull();
        expect(screen.getAllByRole("tab").map((t) => t.getAttribute("data-testid"))).toEqual(["tab-template"]);
        expect(screen.getByRole("button", { name: "Continue" })).toBeEnabled();
      });
    });
  }
}
