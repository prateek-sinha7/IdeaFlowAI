-- the never-dispatched attempts: run, marker seq, events that follow (types), run status/error today
SQL> WITH m AS ( SELECT run_id, seq, created_at, lead(seq) OVER (PARTITION BY run_id ORDER BY seq) AS next_marker_seq FROM run_events WHERE type = 'run_resuming' ), nd AS ( SELECT * FROM m WHERE NOT EXISTS ( SELECT 1 FROM run_events e WHERE e.run_id = m.run_id AND e.seq > m.seq AND (m.next_marker_seq IS NULL OR e.seq < m.next_marker_seq) AND e.type = 'pipeline_start') ) SELECT left(nd.run_id::text, 8) AS run, nd.seq AS marker_seq, nd.created_at::timestamp(0) AS marked_at, (SELECT string_agg(e.seq || ':' || e.type, ', ' ORDER BY e.seq) FROM run_events e WHERE e.run_id = nd.run_id AND e.seq > nd.seq AND (nd.next_marker_seq IS NULL OR e.seq < nd.next_marker_seq)) AS after_marker, w.type AS run_type, w.status AS run_status, left(coalesce(w.error, '<NULL>'), 70) AS run_error FROM nd JOIN workflow_runs w ON w.id = nd.run_id ORDER BY nd.created_at
     ['run', 'marker_seq', 'marked_at', 'after_marker', 'run_type', 'run_status', 'run_error']
