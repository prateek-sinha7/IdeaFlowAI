// stay green under the savedRosterRef fix — (a) spec-017 completion of a fresh ppt_v2 open,
// (b) a realistic ppt_v2 restored 3-of-4 roster survives, (c) a Web/Deck round trip after a
// restore lets the new plan complete the library roster again.

  saveUserWorkflow: vi.fn().mockResolvedValue({ id: "wf-1" }),
  getWorkflowDetail: (...args: unknown[]) => (getWorkflowDetailMock as (...a: unknown[]) => unknown)(...args),
  setWorkflowOverrideEnabled: vi.fn().mockResolvedValue({}),
}));
vi.mock("@/lib/prototype-api", () => ({

import { LaunchWizard } from "./LaunchWizard";

const PPT_V2_PLAN = ["ppt-brief-analyst", "ppt-composer", "ppt-deck-qa-v2", "ppt-code-generator"];
const agent = (id: string, pipeline_type: string, order: number): AgentDef => ({
  id, name: id, role: "r", description: "d", pipeline_type, order, icon: "zap", estimated_duration: 60, has_skill: false,
});
// The real library: ppt_v2 reuses ppt's first two steps, so only 2 of its 4 carry pipeline_type ppt_v2.
const LIB = [
  agent("ppt-brief-analyst", "ppt", 1),
  agent("ppt-composer", "ppt", 2),
  agent("ppt-deck-qa-v2", "ppt_v2", 3),
  agent("ppt-code-generator", "ppt_v2", 4),
];
const preload = (list: AgentDef[]) => ({
  agents: { agents: list, totalCount: list.length, pipelines: {}, status: "succeeded" as const, error: null },
});
const PLAN_V2 = {
  has_override: false, is_overridden: false, override_id: null,
  steps: PPT_V2_PLAN.map((id, i) => ({ agent_id: id, name: id, role: "r", order: i + 1 })),
};
const NO_OVERRIDE = { has_override: false, is_overridden: false, override_id: null, steps: [] };
const pptDraft = () => JSON.parse(sessionStorage.getItem("ppt.draft")!);

beforeEach(() => {
  sessionStorage.clear();
  getWorkflowDetailMock.mockImplementation((_t: unknown, p: unknown) =>
    Promise.resolve(p === "ppt_v2" ? PLAN_V2 : NO_OVERRIDE));
});

describe("QA-KB-048 fixer scratch guards", () => {
  it("(a) spec 017: a fresh ppt_v2 open with no draft completes the 2-agent library roster to the 4-step plan", async () => {
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="ppt_v2" />, { preloadedState: preload(LIB) });
    await screen.findByText("4 agents");
    await user.type(screen.getByLabelText("Brief"), "Pitch deck");
    await user.click(screen.getByTestId("pick-deck-onepager"));
    const cont = screen.getByRole("button", { name: "Continue" });
    await waitFor(() => expect(cont).toBeEnabled());
    await user.click(cont);
    expect(pptDraft().agentIds).toEqual(PPT_V2_PLAN);
  });

  it("(b) a restored ppt_v2 3-of-4 saved roster survives the 4-step plan", async () => {
    sessionStorage.setItem("ppt.draft", JSON.stringify({
      templateId: "onepager", brief: "Pitch deck", agentIds: PPT_V2_PLAN.slice(0, 3),
    }));
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="ppt_v2" />, { preloadedState: preload(LIB) });
    await waitFor(() => expect(getWorkflowDetailMock).toHaveBeenCalled());
    await waitFor(() => expect(screen.getByLabelText("Brief")).toHaveValue("Pitch deck"));
    // let the plan land
    await new Promise((r) => setTimeout(r, 0));
    expect(screen.getByText("3 agents")).toBeInTheDocument();
    const cont = screen.getByRole("button", { name: "Continue" });
    await waitFor(() => expect(cont).toBeEnabled());
    await user.click(cont);
    expect(pptDraft().agentIds).toEqual(PPT_V2_PLAN.slice(0, 3));
  });

  it("(d) a restored ppt_v2 row that is exactly the 2-agent library default is still completed (unsatisfiable alone)", async () => {
    sessionStorage.setItem("ppt.draft", JSON.stringify({
      templateId: "onepager", brief: "Pitch deck", agentIds: ["ppt-deck-qa-v2", "ppt-code-generator"],
    }));
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="ppt_v2" />, { preloadedState: preload(LIB) });
    await waitFor(() => expect(screen.getByLabelText("Brief")).toHaveValue("Pitch deck"));
    await screen.findByText("4 agents");
    const cont = screen.getByRole("button", { name: "Continue" });
    await waitFor(() => expect(cont).toBeEnabled());
    await user.click(cont);
    expect(pptDraft().agentIds).toEqual(PPT_V2_PLAN);
  });

  it("(c) restore, then Web -> Deck: the plan completes the new library roster again", async () => {
    sessionStorage.setItem("ppt.draft", JSON.stringify({
      templateId: "onepager", brief: "Pitch deck", agentIds: PPT_V2_PLAN.slice(0, 3),
    }));
    const user = userEvent.setup();
    renderWithProviders(<LaunchWizard initialMode="ppt_v2" />, { preloadedState: preload(LIB) });
    await waitFor(() => expect(screen.getByLabelText("Brief")).toHaveValue("Pitch deck"));
    await new Promise((r) => setTimeout(r, 0));
    expect(screen.getByText("3 agents")).toBeInTheDocument();
    await user.click(screen.getByTestId("toggle-web"));
    await user.click(screen.getByTestId("toggle-deck"));
    await screen.findByText("4 agents");
    await user.click(screen.getByTestId("pick-deck-onepager"));
    const cont = screen.getByRole("button", { name: "Continue" });
    await waitFor(() => expect(cont).toBeEnabled());
    await user.click(cont);
    expect(pptDraft().agentIds).toEqual(PPT_V2_PLAN);
  });
});
