"""QA-KB-076 fix probe: old (HEAD) guard vs fixed guard on temp inputs. Never touches the real tree."""
import importlib.util, sys, tempfile
from pathlib import Path

def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec); spec.loader.exec_module(mod); return mod

head = load("guard_head", sys.argv[1])
fixed = load("guard_fixed", sys.argv[2])
tmp = Path(tempfile.mkdtemp())

def run_lint(mod, paths):
    mod._SCANNED = paths
    try:
        mod.test_no_line_number_cites_in_shutdown_path_files()
        return "PASSED (guard green)"
    except AssertionError as e:
        return "FAILED -> " + str(e).replace("\n", " | ")

# ISS-1084: a scanned path that no longer exists
gone = tmp / "gone.py"
print("ISS-1084  missing scanned file", gone.name)
print("  HEAD :", run_lint(head, [gone]))
print("  FIXED:", run_lint(fixed, [gone]))
