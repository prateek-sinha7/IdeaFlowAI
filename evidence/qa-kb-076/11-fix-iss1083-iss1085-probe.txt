import importlib.util, shutil, sys, tempfile
from pathlib import Path

def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec); spec.loader.exec_module(mod); return mod

head = load("guard_head", sys.argv[1])
fixed = load("guard_fixed", sys.argv[2])
real_root = fixed._ROOT
tmp_root = Path(tempfile.mkdtemp())
for rel in fixed._ANCHORS:
    (tmp_root / rel).parent.mkdir(parents=True, exist_ok=True)
    shutil.copy(real_root / rel, tmp_root / rel)
rs = tmp_root / "backend/app/api/run_shutdown.py"
rs.write_text(rs.read_text().replace("_drain_then_cancel", "_drain_and_cancel"))
print("temp copy: _drain_then_cancel renamed to _drain_and_cancel in run_shutdown.py")

head._SCANNED = [rs]
