import re, sys
from pathlib import Path

ROOT = Path(sys.argv[1])
API = ROOT / "backend" / "app" / "api"
SCANNED = [
    API / "run_shutdown.py",
    API / "run_engine.py",
    API / "run_stream.py",
    ROOT / "backend" / "app" / "core" / "config.py",
    ROOT / "backend" / "docker-entrypoint.sh",
    ROOT / "backend" / "tests" / "unit" / "services" / "test_shutdown_reachability.py",
]
CURRENT = re.compile(
    r"\b(?:run_commands|engine|model_factory|run_engine|run_stream|run_shutdown|"
    r"authz|websocket|checkpointer|docker-compose)\.(?:py|yml):\d+"
)
# BUGS.md triage sketch
SKETCH = re.compile(r"[\w./-]+\.(?:py|yml|sh) ?:\d+|(?<![\w/.]):\d{3,}")
# evidence/qa-kb-042/09 "broad pattern"
BROAD = re.compile(r"\b[\w./-]+\.(?:py|yml|sh) ?:\d+|(?<![\w/.]):\d{3,}")

for name, rx in (("CURRENT _LINE_CITE", CURRENT), ("BUGS.md SKETCH", SKETCH), ("EVIDENCE-09 BROAD", BROAD)):
    print(f"=== {name}: {rx.pattern}")
    total = 0
    for p in SCANNED:
        if not p.exists():
            print("  MISSING", p); continue
        for n, line in enumerate(p.read_text().splitlines(), 1):
            for m in rx.finditer(line):
                total += 1
                print(f"  {p.name}:{n}: {m.group(0)!r}   | {line.strip()[:110]}")
    print(f"  -> {total} hit(s)\n")
