function fakeFetch(input: unknown): Promise<Response> {
  const url = String(input);
  calls.push(url.replace("http://localhost:8000", ""));
  const u = new URL(url);
  const m = u.pathname.match(/^\/api\/runs\/([^/]+)(\/.*)?$/);
  if (!m) return Promise.resolve(json(404, { detail: "Not Found" }));
  const [, id, rest = ""] = m;
  const ref = u.searchParams.get("ref");
  const files = ref ? LISTING[ref] : LISTING.live;
  if (rest === "/family") return Promise.resolve(json(200, family));
  if (rest === "/sandbox/refs") return Promise.resolve(json(200, { run_id: id, refs: ["v1", "v2", "v3", "v4", "v5", "v7", "v8"] }));
  if (rest === "/sandbox") {
    if (!files) return Promise.resolve(json(400, { detail: "invalid ref" }));
    return Promise.resolve(json(200, { run_id: id, expired: false, truncated: false, ...(ref ? { ref } : {}), files: files.map(entry) }));
  }
  if (rest === "/sandbox/file") {
    if (!files) return Promise.resolve(json(400, { detail: "invalid ref" }));
    const path = u.searchParams.get("path")!;
    if (!files.includes(path)) return Promise.resolve(json(404, { detail: "File not found" }));
    return Promise.resolve(new Response(path.endsWith(".pdf") ? "%PDF-1.4 scratch bytes" : "# backlog text", { status: 200 }));
  }
  if (rest === "") {
    const mem = family.members.find((x) => x.id === id)!;
    return Promise.resolve(json(200, {
      id, title: mem.title, type: "backlog", status: mem.status, input: "Write the complete product backlog", output: "# Backlog text form",
      agent_outputs: "[]", deliverable_mimetype: "application/pdf", deliverable_filename: "Backlog.pdf",
      parent_run_id: mem.parent_run_id, root_run_id: family.root_id, created_at: mem.created_at, completed_at: mem.completed_at, agent_count: 0,
    }));
  }
  return Promise.resolve(json(404, { detail: "Not Found" }));
}

// PROTOTYPE OF THE PROPOSED FIX, in the scratchpad only (not application source): SandboxPdfPreview resolves through the shared
// resolver first, reads the RESOLVED path at the ref, and renders `fallback` WITHOUT a file request when nothing resolves.
function ProposedSandboxPdfPreview({ runId, path, selectedRef, fallback }: { runId: string; path: string; selectedRef?: string | null; fallback: ReactNode }) {
  const [state, setState] = useState<{ url: string } | "loading" | "missing">("loading");
  useEffect(() => {
    if (selectedRef === null) return;
    let url: string | null = null;
    let cancelled = false;
    setState("loading");
    void (async () => {
      try {
        const token = getToken() || "";
        const hit = await resolveRunDeliverable(token, runId, path, selectedRef);
        if (cancelled) return;
        if (!hit) { setState("missing"); return; }
        const blob = await getRunSandboxFileBlob(token, runId, hit.path, hit.ref ?? selectedRef);
        if (cancelled) return;
        url = URL.createObjectURL(new Blob([blob], { type: "application/pdf" }));
        setState({ url });
      } catch { if (!cancelled) setState("missing"); }
    })();
    return () => { cancelled = true; if (url) URL.revokeObjectURL(url); };
  }, [runId, path, selectedRef]);
  if (state === "missing" || selectedRef === null) return <>{fallback}</>;
  if (state === "loading") return <div data-testid="sandbox-pdf-loading" />;
  return <iframe title={path} src={state.url} data-testid="sandbox-pdf-preview" />;
}
