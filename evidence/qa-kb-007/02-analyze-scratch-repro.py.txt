def _seed_run(session, *, run_id="run-1", owner_id="alice", workspace_id="ws-1"):

            id=run_id, user_id=owner_id, title="t", type="prototype",

def _stored(session, event_id):
    """Raw SQL read: what is ACTUALLY in the table, bypassing any ORM identity map."""
    import json

    session.expire_all()
    seq, payload = session.execute(
        text("SELECT seq, payload_json FROM run_events WHERE event_id = :e"), {"e": event_id}
    ).one()
    payload = json.loads(payload) if isinstance(payload, str) else payload
    return seq, payload

# ───────────── Part 2: model the agreed minimal fix on a patched COPY ─────────────


class PatchedStore(ScopedStore):
    """Scratch-only copy of append_event_at_or_after + the BUGS.md agreed fix (3 lines)."""

    async def append_event_at_or_after(self, run_id, seq, event_id, type, payload_json,
                                       _max_attempts=8):
        attempt_seq = int(seq)
        last_rejection = None
        for _ in range(max(1, _max_attempts)):
            try:
                await self.append_event(run_id, attempt_seq, event_id, type, payload_json)
                return attempt_seq
            except IntegrityError as rejected:
                last_rejection = rejected
                if self._session is not None:
                    self._session.rollback()
                tail = await self._max_event_seq(run_id)
                if tail < attempt_seq:
                    raise
                attempt_seq = tail + 1
                # ---- the agreed minimal fix -------------------------------------------
                if isinstance(payload_json, dict) and "seq" in payload_json:
                    payload_json = {**payload_json, "seq": attempt_seq}
                # -----------------------------------------------------------------------
        raise last_rejection


@pytest.mark.asyncio
async def test_D_fix_single_hop(db_session):
    _seed_run(db_session)
    store = PatchedStore(owner_id="alice", workspace_id="ws-1", session=db_session)
    await store.append_event("run-1", 1, "a", "t", {"seq": 1})
    landed = await store.append_event_at_or_after("run-1", 1, "b", "t", {"seq": 1})
    row_seq, payload = _stored(db_session, "b")
    assert landed == row_seq == payload["seq"] == 2


@pytest.mark.asyncio
async def test_E_fix_multi_hop_restamps_every_retry(db_session, monkeypatch):
    """Two successive collisions: the FINAL attempt's seq must be what is stored."""
    _seed_run(db_session)
    store = PatchedStore(owner_id="alice", workspace_id="ws-1", session=db_session)
    for s in (1, 2, 3):
        await store.append_event("run-1", s, f"x{s}", "t", {"seq": s})
    real_max = ScopedStore._max_event_seq
    probes = {"n": 0}

    async def _stale_first_probe(self, rid):
        probes["n"] += 1
        return 1 if probes["n"] == 1 else await real_max(self, rid)   # first probe: stale tail

    monkeypatch.setattr(ScopedStore, "_max_event_seq", _stale_first_probe)
    landed = await store.append_event_at_or_after("run-1", 1, "b", "t", {"seq": 1})
    row_seq, payload = _stored(db_session, "b")
    print(f"\n[E] hops={probes['n']} landed={landed} row.seq={row_seq} payload.seq={payload['seq']}")
    assert probes["n"] == 2                      # two collisions -> two tail probes
    assert landed == row_seq == payload["seq"] == 4


@pytest.mark.asyncio
async def test_F_fix_leaves_seqless_payloads_and_the_callers_dict_untouched(db_session):
    _seed_run(db_session)
    store = PatchedStore(owner_id="alice", workspace_id="ws-1", session=db_session)
    await store.append_event("run-1", 1, "a", "t", {"seq": 1})
    marker = {"pipeline_run_id": "run-1", "reason": "r"}          # resume-marker / cancel shape
    await store.append_event_at_or_after("run-1", 1, "m", "run_resuming", marker)
    _, stored = _stored(db_session, "m")
    assert stored == {"pipeline_run_id": "run-1", "reason": "r"} and "seq" not in marker
    # caller's own dict is NOT mutated (it is the live frame object; the caller re-stamps it)
    mine = {"seq": 1, "event_id": "c"}
    landed = await store.append_event_at_or_after("run-1", 1, "c", "t", mine)
    assert landed == 3 and mine["seq"] == 1


@pytest.mark.asyncio
async def test_G_fix_exhaustion_leaves_callers_frame_seq_alone(db_session, monkeypatch):
    """All attempts rejected -> raises; with a COPY the live frame keeps its requested seq
    (the persist() 'degraded => no re-stamp' contract at engine.py:4203)."""
    _seed_run(db_session)
    store = PatchedStore(owner_id="alice", workspace_id="ws-1", session=db_session)
    await store.append_event("run-1", 1, "a", "t", {"seq": 1})
    frame = {"seq": 1, "event_id": "z"}

    async def _always_tail_ahead(self, rid):
        return 10**6                      # every probe says "tail reached" -> keeps retrying

    monkeypatch.setattr(ScopedStore, "_max_event_seq", _always_tail_ahead)
    # also force every append to be rejected
    async def _reject(self, *a, **k):
        raise IntegrityError("x", None, Exception("forced"))
    monkeypatch.setattr(ScopedStore, "append_event", _reject)
    with pytest.raises(IntegrityError):
        await store.append_event_at_or_after("run-1", 1, "z", "t", frame, _max_attempts=3)
    assert frame["seq"] == 1
