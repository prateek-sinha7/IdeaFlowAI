def fresh_session():
    eng = create_engine(
        "sqlite://", connect_args={"check_same_thread": False}, poolclass=StaticPool
    )
    Base.metadata.create_all(bind=eng)
    s = sessionmaker(bind=eng, autocommit=False, autoflush=False)()
    s.add(WorkflowRun(id="run-1", user_id="alice", title="t", type="prototype",
                      status="running", input="idea", owner_id="alice",
                      workspace_id="ws-1"))
    s.commit()
    return s


def stored(s, event_id):
    s.expire_all()
    r = s.execute(text("SELECT seq, payload_json FROM run_events WHERE event_id = :e"),
                  {"e": event_id}).one_or_none()
    if r is None:
        return None
    p = json.loads(r[1]) if isinstance(r[1], str) else r[1]
    return r[0], p

import os

import pytest


def _mutant(kind):
    async def at_or_after(self, run_id, seq, event_id, type, payload_json, _max_attempts=8):
        from sqlalchemy.exc import IntegrityError

        attempt_seq, last, restamped = int(seq), None, False
        for _ in range(max(1, _max_attempts)):
            try:
                await self.append_event(run_id, attempt_seq, event_id, type, payload_json)
                return attempt_seq
            except IntegrityError as rejected:
                last = rejected
                if self._session is not None:
                    self._session.rollback()
                tail = await self._max_event_seq(run_id)
                if tail < attempt_seq:
                    raise
                attempt_seq = tail + 1
                if isinstance(payload_json, dict) and "seq" in payload_json:
                    if kind == "A":
                        payload_json["seq"] = attempt_seq
                    elif not restamped:
                        payload_json = {**payload_json, "seq": attempt_seq}
                        restamped = True
        raise last

    return at_or_after


def install(kind):
    from agents.authz import ScopedStore

    ScopedStore.append_event_at_or_after = _mutant(kind)


@pytest.hookimpl(tryfirst=True)
def pytest_runtest_setup(item):
    install(os.environ["KB007_MUTANT"])
