function initialWorkflowTypeFor(parsed: ParsedView): string | undefined {
  switch (parsed.screen) {
    case "create-app":
      return "app_builder";
    case "create-user-stories":
      return "user_stories";
    case "create-workflow":
      return parsed.pipelineType;
    default:
      return undefined;
  }
}
