// SCRATCH (outside the repo), run with vitest.head.config.ts = the PRE-FIX components from git HEAD.
// Part A — CONTROL on the path everyone agrees works today (focused click, like Chrome):
//   records state RIGHT AFTER Escape, i.e. what the escape tests' synchronous null-check sees.
// Part B — the escape tests' bodies as they will read after the waitFor correction, unfocused click:
//   plain `it`, so they must FAIL here (bug present) for the corrected tests to still discriminate.
import { describe, expect, it, vi, beforeEach } from "vitest";
import { render, screen, fireEvent, waitFor } from "@testing-library/react";
import { renderWithProviders } from "@/test/renderWithProviders";
import userEvent from "@testing-library/user-event";
import React, { useState } from "react";

const { routerObj, api } = vi.hoisted(() => ({
  routerObj: { replace: vi.fn(), push: vi.fn(), prefetch: vi.fn() },
  api: {
    getToken: vi.fn(() => "tok"), getMe: vi.fn(), logout: vi.fn(), adminListUsers: vi.fn(),
    adminUpdateTier: vi.fn(), adminCreateUser: vi.fn(), adminDeleteUser: vi.fn(), adminUpdateRole: vi.fn(),
    adminDisableUser: vi.fn(), adminEnableUser: vi.fn(), onAdminMfaRequired: vi.fn(() => () => {}),
  },
}));
vi.mock("next/navigation", () => ({ useRouter: () => routerObj }));
vi.mock("@/lib/api", () => api);
vi.mock("@/hooks/useWorkflowMetadata", () => ({
  useWorkflowShortNames: () => () => undefined,
  useWorkflowLabels: () => () => undefined,
}));

import { NotificationPanel } from "@/components/ui/NotificationPanel";
import AdminPage from "@/app/admin/page";
import { FamilyGroupCard, type FamilyGroup } from "@/components/history/RevisionFamilyView";
import type { WorkflowRun } from "@/types/index";

const USERS = [{
  id: "aaaa1111-2222-3333-4444-555566667777", email: "alice@example.com", tier: "pro" as const,
  is_admin: false, status: "active" as const, created_at: "2026-01-15T09:00:00Z", workflow_run_count: 12,
}];
beforeEach(() => {
  vi.clearAllMocks();
  api.getToken.mockReturnValue("tok");
  api.getMe.mockResolvedValue({ is_admin: true, email: "admin@example.com" });
  api.adminListUsers.mockResolvedValue(USERS);
});
const run = {
  id: "b0642e0f-fe74-4f7e-9f16-8b164a6e989d", title: "A history run", type: "custom",
  status: "completed", createdAt: "2026-09-01T14:46:55.000Z", agentCount: 3,
} as WorkflowRun;
const group = { root: run, members: [run], latest: run } as FamilyGroup;
function Harness() {
  const [openMenuId, setOpenMenuId] = useState<string | null>(null);
  return (
    <FamilyGroupCard group={group} index={0} expanded={false} onToggle={() => {}} onSelectRun={() => {}}
      openMenuId={openMenuId}
      onToggleMenu={(id, e) => { e?.stopPropagation(); setOpenMenuId(openMenuId === id ? null : id); }}
      onDeleteClick={() => {}} />
  );
}
const bellUi = () => (
  <NotificationPanel notifications={[]} unreadCount={0} onMarkAllRead={vi.fn()} onClearAll={vi.fn()}
    onGoToPipeline={vi.fn()} onViewResults={vi.fn()} />
);

describe("A: pre-fix, FOCUSED click (working path) — state right after Escape", () => {
  it("bell", async () => {
    const user = userEvent.setup();
    renderWithProviders(bellUi());
    const bell = screen.getByRole("button", { name: /notifications/i });
    await user.click(bell);
    await user.keyboard("{Escape}");
    console.log("[A bell] aria-expanded =", bell.getAttribute("aria-expanded"), "| dialog still in DOM =", !!screen.queryByRole("dialog"));
  });
  it("admin tier", async () => {
    render(<AdminPage />);
    await screen.findByText("alice@example.com");
    const user = userEvent.setup();
    const trigger = document.querySelector('[aria-haspopup="menu"]') as HTMLElement;
    await user.click(trigger);
    await user.keyboard("{Escape}");
    console.log("[A admin] aria-expanded =", trigger.getAttribute("aria-expanded"), "| menu still in DOM =", !!screen.queryByRole("menu"));
  });
  it("history RowMenu", async () => {
    const user = userEvent.setup();
    render(<Harness />);
    const trigger = screen.getByRole("button", { name: "Run actions" });
    await user.click(trigger);
    await user.keyboard("{Escape}");
    console.log("[A rowmenu] aria-expanded =", trigger.getAttribute("aria-expanded"), "| menu still in DOM =", !!screen.queryByRole("menu"));
  });
});

describe("B: pre-fix, UNFOCUSED click — corrected test bodies (must FAIL here)", () => {
  it("bell (corrected NotificationPanel.escape body)", async () => {
    const user = userEvent.setup();
    renderWithProviders(bellUi());
    const bell = screen.getByRole("button", { name: /notifications/i });
    fireEvent.click(bell);
    expect(bell).toHaveAttribute("aria-expanded", "true");
    await user.keyboard("{Escape}");
    await waitFor(() => expect(screen.queryByRole("dialog")).toBeNull());
    expect(bell).toHaveAttribute("aria-expanded", "false");
  });
  it("admin tier (corrected admin.escape body)", async () => {
    render(<AdminPage />);
    await screen.findByText("alice@example.com");
    const user = userEvent.setup();
    const trigger = document.querySelector('[aria-haspopup="menu"]') as HTMLElement;
    fireEvent.click(trigger);
    expect(trigger).toHaveAttribute("aria-expanded", "true");
    await user.keyboard("{Escape}");
    await waitFor(() => expect(screen.queryByRole("menu")).toBeNull());
    expect(trigger).toHaveAttribute("aria-expanded", "false");
  });
  it("history RowMenu (corrected RevisionFamilyView.escape body)", async () => {
    const user = userEvent.setup();
    render(<Harness />);
    const trigger = screen.getByRole("button", { name: "Run actions" });
    fireEvent.click(trigger);
    expect(trigger).toHaveAttribute("aria-expanded", "true");
    await user.keyboard("{Escape}");
    expect(trigger).toHaveAttribute("aria-expanded", "false");
    await waitFor(() => expect(screen.queryByRole("menu")).toBeNull());
  });
});
