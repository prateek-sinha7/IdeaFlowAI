## gate_off_plugin.py (scratch, not in the repo)
"""Scratch probe: disable _validate_ref (pass-through) to prove a test depends on the ref gate.
Imports app lazily in pytest_runtest_setup so collection never binds a live DB."""
import pytest

@pytest.hookimpl(tryfirst=True)
def pytest_runtest_setup(item):
    from app.api import run_files as rf
    mp = pytest.MonkeyPatch()
    mp.setattr(rf, "_validate_ref", lambda ref, sandbox: ref)
    item._gate_off_mp = mp

def pytest_runtest_teardown(item):
    mp = getattr(item, "_gate_off_mp", None)
    if mp is not None:
        mp.undo()
