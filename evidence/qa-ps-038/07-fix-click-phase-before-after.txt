    from framework.locators import run_history as RH

    INIT = """
    window.__runsCalls = [];
    const _f = window.fetch;
    window.fetch = function (...args) {
      const url = typeof args[0] === 'string' ? args[0] : (args[0] && args[0].url) || '';
      if (url.includes('/api/runs?')) {
        window.__runsCalls.push({ url, stack: new Error().stack });
      }
      return _f.apply(this, args);
    };
    """

    def test_probe(page):
        page.add_init_script(INIT)
        page.goto("/runs")
        from playwright.sync_api import expect
        expect(page.locator(RH.ROW).first).to_be_visible(timeout=20000)
        page.wait_for_timeout(2500)
        page.evaluate("window.__runsCalls = []")
        rows = page.locator(RH.ROW).evaluate_all(
            "els => els.map((e) => e.getAttribute('aria-label'))")
        done = [i for i, l in enumerate(rows) if RH.status_of(l) == "completed"]
        page.locator(RH.ROW).nth(done[0]).click()
        page.wait_for_url(lambda url: "/runs/" in url)
        page.wait_for_timeout(5000)
        for c in page.evaluate("window.__runsCalls"):
            print(c["url"], c["stack"])
