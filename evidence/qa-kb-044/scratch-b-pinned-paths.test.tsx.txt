import { describe, expect, it, vi, beforeEach } from "vitest";
import { fireEvent, render, screen, waitFor } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import React from "react";

const mockGetToken = vi.fn(() => "test-token");
const mockGetWorkflow = vi.fn<(token: string, id: string) => Promise<any>>();
const mockGetRunSandboxRefs = vi.fn();
vi.mock("@/lib/api", () => ({
  getToken: () => mockGetToken(),
  getWorkflow: (token: string, id: string) => mockGetWorkflow(token, id),
  getRunFamily: vi.fn(),
  postShareRun: vi.fn(),
  deleteShareRun: vi.fn(),
  getRunSandboxRefs: (...a: unknown[]) => mockGetRunSandboxRefs(...a),
  getRunSandbox: vi.fn().mockResolvedValue({ run_id: "x", expired: false, truncated: false, files: [] }),
  getRunSandboxFile: vi.fn().mockResolvedValue(""),
  getRunSandboxFileBlob: vi.fn(),
  getRunSandboxZip: vi.fn(),
  resolveRunDeliverable: vi.fn().mockResolvedValue(null),
}));
const STRIPPED_MOTION_PROPS = new Set(["initial","animate","exit","transition","whileHover","whileTap","whileFocus","whileInView","viewport","layout","layoutId","drag","dragConstraints","variants","custom"]);
vi.mock("motion/react", () => ({
  motion: new Proxy({}, { get: (_t, prop: string) => ({ children, ...rest }: any) => React.createElement(prop, Object.fromEntries(Object.entries(rest).filter(([k]) => !STRIPPED_MOTION_PROPS.has(k))), children) }),
  AnimatePresence: ({ children }: any) => <>{children}</>,
}));
vi.mock("/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/UserStoryPreview", () => ({ UserStoryPreview: () => <div data-testid="user-story-preview" /> }));
vi.mock("/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/PPTPreview", () => ({ PPTPreview: () => <div data-testid="ppt-preview" /> }));
vi.mock("/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/PrototypePreview", () => ({ PrototypePreview: () => <div data-testid="proto-preview" /> }));
vi.mock("/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/MarkdownPreview", () => ({ MarkdownPreview: () => <div data-testid="markdown-preview" /> }));
vi.mock("/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/AppBuilderPreview", () => ({ AppBuilderPreview: () => <div data-testid="appbuilder-preview" /> }));
vi.mock("@/components/results/FilesTab", () => ({ FilesTab: () => <div data-testid="files-tab" />, downloadBlob: vi.fn(), deriveDeliverableFilename: () => "prototype.html" }));
vi.mock("@/components/results/AgentThinkingTab", () => ({ AgentThinkingTab: () => <div data-testid="thinking-tab" /> }));
vi.mock("@/components/results/AuditTab", () => ({ AuditTab: () => <div data-testid="audit-tab" /> }));

import { PreviewPanel } from "/Users/bilala/Developer/Projects/VELOCITY-AI-worktrees/feat-user-stories-v2/frontend/src/components/preview/PreviewPanel";

const t0 = new Date("2026-08-13T10:00:00Z").toISOString();
const t1 = new Date("2026-08-13T11:00:00Z").toISOString();
const tagless: any = {
  root_id: "root",
  members: [
    { id: "root", type: "prototype", title: "v1", status: "completed", revision_index: 0, parent_run_id: null, created_at: t0, completed_at: t0 },
    { id: "r1", type: "prototype_revision", title: "v2", status: "completed", revision_index: 1, parent_run_id: "root", created_at: t1, completed_at: t1 },
  ],
};
const tagged: any = {
  root_id: "root",
  members: [
    { ...tagless.members[0], git_tag: "v1" },
    { ...tagless.members[1], git_tag: "v2" },
  ],
};
beforeEach(() => {
  mockGetToken.mockReset().mockReturnValue("test-token");
  mockGetWorkflow.mockReset();
  mockGetRunSandboxRefs.mockReset().mockResolvedValue([]);
});
const NEUTRAL = "Output will appear here";

describe("B - which paths reach the !hasContent placeholder for a pinned older version", () => {
  it("B1. DEEP-LINK pin (pinnedVersion prop), tagless v1, completed, output=null -> placeholder shown (BUG REAL)", async () => {
    mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, status: "completed", output: null } as any));
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagless} liveRunId="root" pinnedVersion="1" />);
    await waitFor(() => expect(mockGetWorkflow).toHaveBeenCalledWith("test-token", "root"));
    await screen.findByRole("status"); // read-only banner => viewingVersion is set
    expect(screen.getByText(NEUTRAL)).toBeInTheDocument();
  });

  it("B2. same pin, but the pinned run is still RUNNING (non-terminal) -> same placeholder (a status-blind 'no deliverable saved' copy would be false here)", async () => {
    mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, status: "running", output: null } as any));
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagless} liveRunId="root" pinnedVersion="1" />);
    await waitFor(() => expect(mockGetWorkflow).toHaveBeenCalledWith("test-token", "root"));
    await screen.findByRole("status");
    expect(screen.getByText(NEUTRAL)).toBeInTheDocument();
  });

  it("B3. pinned run FAILED with output=null -> still the neutral placeholder (shell-keyed failure affordance does not see the pinned status)", async () => {
    mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, status: "failed", output: null } as any));
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagless} liveRunId="root" pinnedVersion="1" />);
    await waitFor(() => expect(mockGetWorkflow).toHaveBeenCalledWith("test-token", "root"));
    await screen.findByRole("status");
    expect(screen.getByText(NEUTRAL)).toBeInTheDocument();
    expect(screen.queryByText(/did not complete successfully/i)).toBeNull();
  });

  it("B4. CONTROL: pinned version WITH output renders the deliverable, no placeholder", async () => {
    mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, status: "completed", output: "<html>v1 body</html>" } as any));
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagless} liveRunId="root" pinnedVersion="1" />);
    await waitFor(() => expect(mockGetWorkflow).toHaveBeenCalledWith("test-token", "root"));
    await screen.findByRole("status");
    expect(screen.queryByText(NEUTRAL)).toBeNull();
    expect(screen.getByTestId("proto-preview")).toBeInTheDocument();
  });

  it("B5. CONTROL: no pin, terminal, no content -> neutral copy stays (PreviewPanel.degraded.test.tsx:177/259 + TS-Q-05 pin this)", () => {
    render(<PreviewPanel workflowType="prototype" isStreaming={false} />);
    expect(screen.getByText(NEUTRAL)).toBeInTheDocument();
  });

  it("B6. VERSION-MENU path with git tags: pick tag v1 -> RefVersionPreview branch, NOT the placeholder, even though getWorkflow output=null", async () => {
    mockGetRunSandboxRefs.mockResolvedValue(["v1", "v2"]);
    mockGetWorkflow.mockImplementation((_t, id) => Promise.resolve({ id, status: "completed", output: null } as any));
    const user = userEvent.setup();
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagged} liveRunId="r1" />);
    // chip becomes the interactive VersionMenu once tags load
    const chip = await screen.findByRole("button", { name: /v2/ });
    await user.click(chip);
    await user.click(await screen.findByRole("option", { name: "v1" }));
    await waitFor(() => expect(mockGetWorkflow).toHaveBeenCalledWith("test-token", "root"));
    await screen.findAllByRole("status");
    expect(await screen.findAllByText(/Viewing v1 · read-only/)).not.toHaveLength(0); // RefVersionPreview banner => the ref branch is what rendered
    expect(screen.queryByText(NEUTRAL)).toBeNull();
  });

  it("B7. VERSION MENU with NO tags: no interactive picker is mounted at all (static chip)", async () => {
    mockGetRunSandboxRefs.mockResolvedValue([]);
    render(<PreviewPanel workflowType="prototype" prototypeContent="<html>latest</html>" runFamily={tagless} liveRunId="r1" />);
    await waitFor(() => expect(mockGetRunSandboxRefs).toHaveBeenCalled());
    expect(screen.queryByRole("listbox")).toBeNull();
    expect(screen.queryByRole("button", { name: /v2/ })).toBeNull();
    expect(screen.getByText("v2").tagName).toBe("SPAN"); // static, non-interactive chip
  });
});
