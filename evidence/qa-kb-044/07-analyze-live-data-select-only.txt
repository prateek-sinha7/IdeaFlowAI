-- Q2  multi-member families that contain a NULL/empty-output terminal member (members in created_at order)
WITH RECURSIVE fam AS (
  SELECT id, id AS root_id FROM workflow_runs WHERE parent_run_id IS NULL
  UNION ALL
  SELECT c.id, f.root_id FROM workflow_runs c JOIN fam f ON c.parent_run_id = f.id
), multi AS (SELECT root_id FROM fam GROUP BY root_id HAVING count(*) > 1),
nullfam AS (SELECT DISTINCT f.root_id FROM fam f JOIN workflow_runs r ON r.id = f.id
            WHERE r.status IN ('completed','failed','cancelled','degraded','diverted') AND (r.output IS NULL OR r.output = ''))
SELECT f.root_id, r.id, r.type, r.status, to_char(r.created_at,'MM-DD HH24:MI'), (r.output IS NULL) AS out_null, (r.git_tag IS NOT NULL) AS tagged
FROM fam f JOIN workflow_runs r ON r.id = f.id
WHERE f.root_id IN (SELECT root_id FROM multi) AND f.root_id IN (SELECT root_id FROM nullfam)
ORDER BY f.root_id, r.created_at
