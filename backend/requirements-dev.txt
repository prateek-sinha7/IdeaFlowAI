# Testing framework
#
# pytest 8.3.4 carried PYSEC-2026-1845; 9.0.3 is the fix. That bump forces two
# companions forward as a set, so all three move together:
#   - pytest-asyncio 0.24.0 declares `pytest>=8.2,<9` and would hard-conflict.
#     1.4.0 declares `pytest>=8.4,<10`. It still honours the
#     `asyncio_default_fixture_loop_scope` setting in pyproject.toml.
#   - pytest-cov 5.0.0 predates pytest 9; 7.1.0 is the current line.
pytest==9.0.3
pytest-asyncio==1.4.0
hypothesis==6.122.3

# HTTP testing client

# Security scanning — SAST + dependency CVE audit.
#
# ci.yml's `backend` job invokes both via `uv run --no-sync bandit ...` /
# `uv run --no-sync pip-audit ...` against the venv `uv sync --frozen --group
# dev` installed from the committed backend/uv.lock (see
# .github/actions/setup-backend/action.yml) — declaring them in the `dev`
# dependency group (mirrored here) is what makes those tools present.
#
# The [toml] extra is what lets `bandit -c pyproject.toml` read
# [tool.bandit] from backend/pyproject.toml, which is exactly how ci.yml
# invokes it.
bandit[toml]==1.9.4
pip-audit==2.10.1

# Hardened XML parsing for the CI report readers (agents/ci/*).
#
# Those modules read coverage.xml and JUnit XML, and bandit B314 blocks
# xml.etree.ElementTree.parse/fromstring outright; the documented remediation
# is the defusedxml equivalent rather than a `# nosec` suppression. defusedxml
# was already present in uv.lock as a transitive dependency of
# pip-audit -> cyclonedx-python-lib -> py-serializable, so this pin turns an
# incidental edge into a declared one that a pip-audit bump cannot silently
# remove. Dev-only: nothing under app/ imports agents.ci.
defusedxml==0.7.1
