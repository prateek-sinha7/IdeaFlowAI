  - id: string (e.g., "WO-2026-0515")
  - vehicleId: string (FK to Vehicle id)
  - shop: string (Downtown Bay | North Warehouse | East Terminal)
  - task: string (e.g., "Engine Oil Change")

  - estimatedCompletion: date (ISO)
  - laborHours: number (e.g., 1.5)
  - laborCost: number (USD, e.g., 180.00)
  - lineItems: array of { partId, partName, qty, unitCost, totalCost }
  - partsCost: number (sum of lineItems totals)
  - totalCost: number (laborCost + partsCost)
  - notes: text (technician notes, e.g., "Brake wear evident, recommend replacement.")

  - id: string (e.g., "INS-2026-0518")

  - date: date (ISO)
  - technician: string (name, e.g., "Mike R.")
  - findings: text (detailed findings, e.g., "Clean, no issues noted")
  - status: enum (Passed | Failed | Pending | N/A)

  - correctiveActions: array of { action, completedDate } (actions taken post-inspection)

  - nextDueMileage: number (estimated)

    // Handle route changes
    function handleRouteChange() {
      const hash = window.location.hash.replace(/^#\/?/, '') || '';
      let pageId = 'vehicles';
      let vehicleId = null;

      // Parse hash to determine page and parameters
      if (!hash || hash === '') {
        pageId = 'vehicles';
      } else if (hash.startsWith('vehicle/')) {
        pageId = 'vehicle';
        vehicleId = hash.replace('vehicle/', '');
        store.currentVehicleId = vehicleId;
      } else if (hash === 'work-orders') {
        pageId = 'work-orders';
      } else if (hash === 'inspections') {
        pageId = 'inspections';
      } else if (hash === 'parts-costs') {
        pageId = 'parts-costs';
      } else if (hash === 'settings') {
        pageId = 'settings';
      }

      store.currentPage = pageId;

      // Hide all pages
      document.querySelectorAll('section[data-page]').forEach(section => {
        section.classList.remove('is-active');
      });

      // Show active page
      const activePage = document.querySelector(`section[data-page="${pageId}"]`);
      if (activePage) {
        activePage.classList.add('is-active');
      }

      // Update sidebar active nav item
      document.querySelectorAll('.nav-item').forEach(link => {
        link.classList.remove('nav-item--active');
      });
      // For vehicle detail, keep the vehicles nav link active
      const navPageId = pageId === 'vehicle' ? 'vehicles' : pageId;
      document.querySelectorAll(`[data-page-link="${navPageId}"]`).forEach(link => {
        link.classList.add('nav-item--active');
      });

      // Update topbar title
      const title = pageTitles[pageId] || 'Dashboard';
      document.getElementById('topbarTitle').textContent = title;

      // For vehicle detail, show vehicle ID in title (placeholder)
      if (pageId === 'vehicle' && vehicleId) {
        document.getElementById('vehicleDetailTitle').textContent = `Vehicle: ${vehicleId.toUpperCase()}`;
      }

      // Initialize inspection handlers when navigating to inspections page
      if (pageId === 'inspections') {
        setTimeout(() => {
          initInspectionHandlers();
        }, 0);
      }

      // Render work orders when navigating to work-orders page
      if (pageId === 'work-orders') {
        setTimeout(() => {
          renderWorkOrders();
        }, 0);
      }
    }
