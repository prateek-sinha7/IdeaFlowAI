## Navigation Flows

**Navigation Map**:
1. **Vehicles page** (`#/vehicles`) → [View] button on any vehicle row → **Vehicle Detail** (`#/vehicle/:id`)
2. **Vehicle Detail** (`#/vehicle/:id`) → [Advance Status] button → modal → on confirm, vehicle status updates globally, immediately visible on Vehicles and Work Orders pages
3. **Vehicle Detail** → [Flag for Parts] button → modal → on submit, creates/updates line item in active work order, updates Parts & Costs totals
4. **Vehicle Detail** → [View Work Orders] link → **Work Orders** (`#/work-orders`) filtered to this vehicle
5. **Vehicle Detail** → breadcrumb "Vehicles" or [← Back to Vehicles] → back to **Vehicles** page
6. **Work Orders** (`#/work-orders`) → drag vehicle card between Kanban columns → vehicle status transitions, reflected on Vehicles page immediately
7. **Work Orders** → click vehicle card → modal with work order details, line items, costs
8. **Work Orders** → [+ New Work Order] → modal form to create new work order
9. **Work Orders** → shop selector dropdown → filters displayed work orders to selected shop
10. **Inspections** (`#/inspections`) → [Assign] button → modal to assign inspector
11. **Inspections** → [View] button → modal with inspection details and findings
12. **Inspections** → [+ Schedule Inspection] → modal form to schedule new inspection
13. **Parts & Costs** (`#/parts-costs`) → [View] link in work orders column → navigates to **Work Orders** page, filtered to those work orders
14. **Settings** (`#/settings`) → [Save] buttons persist settings globally
15. **Sidebar navigation** → click any nav item (Vehicles, Work Orders, Inspections, Parts & Costs, Settings) → navigate to that page

**Persistent Global State Cascades**:
- Vehicle status advance in Vehicle Detail immediately updates Vehicles table and Work Orders Kanban
- Drag-drop status transitions in Work Orders Kanban immediately update vehicle status on Vehicles page
- Work order line items added in Vehicle Detail immediately aggregate to Parts & Costs page totals
- Shop selector on Work Orders filters all subsequent work order views; persists across page navigation
- Settings changes persist globally across all pages

    function navigateTo(pageId, param = null) {
      if (param) {
        window.location.hash = `#/${pageId}/${param}`;
      } else {
        window.location.hash = `#/${pageId}`;
      }
    }

    function parseRoute() {
      const hash = window.location.hash.replace(/^#\/?/, '') || 'vehicles';
      const parts = hash.split('/');
      return {
        page: parts[0],
        param: parts[1] || null
      };
    }

    function handleRouteChange() {
      const route = parseRoute();
      const pageId = route.page;
      const param = route.param;

      // Hide all pages
      document.querySelectorAll('section[data-page]').forEach(s => s.classList.remove('is-active'));

      // Show current page
      let targetPage = null;
      if (pageId === 'vehicle' && param) {
        targetPage = document.querySelector('section[data-page="vehicle-detail"]');
      } else {
        targetPage = document.querySelector(`section[data-page="${pageId}"]`);
      }

      if (targetPage) {
        targetPage.classList.add('is-active');
      }

      // Update nav active state
      document.querySelectorAll('a.nav-item').forEach(a => a.classList.remove('active'));
      document.querySelectorAll(`a.nav-item[data-page-link="${pageId}"]`).forEach(a => a.classList.add('active'));

      // Update topbar title based on page
      const titleMap = {
        vehicles: 'Vehicles',
        vehicle: 'Vehicle Detail',
        'work-orders': 'Work Orders',
        inspections: 'Inspections',
        'parts-costs': 'Parts & Costs',
        settings: 'Settings'
      };
      const titleElement = document.querySelector('[data-od-id="topbar-title"]');
      if (titleElement) {
        titleElement.textContent = titleMap[pageId] || 'Dashboard';
      }

      // Initialize page-specific content
      if (pageId === 'vehicles') {
        initializeVehicleTable();
      } else if (pageId === 'vehicle' && param) {
        initializeVehicleDetail();
      } else if (pageId === 'work-orders') {
        initializeWorkOrders();
      } else if (pageId === 'inspections') {
        initializeInspectionsPage();
      } else if (pageId === 'parts-costs') {
        initializePartsCosts();
      } else if (pageId === 'settings') {
        initializeSettings();
      }
    }

    // Get status badge class
    function getStatusClass(status) {
      const statusMap = {
        'Available': 'available',
        'In Shop': 'in-shop',
        'Awaiting Parts': 'awaiting',
        'Road Test': 'test',
        'Out of Service': 'out',
        'Complete': 'complete'
      };
      return statusMap[status] || 'available';
    }

    // Filter and render vehicle table
    function renderVehicleTable() {
      const filters = appState.uiState.filters;
      let filtered = appState.vehicles;

      // Apply type filter
      if (filters.typeFilters.length > 0) {
        filtered = filtered.filter(v => filters.typeFilters.includes(v.type));
      }

      // Apply status filter
      if (filters.statusFilter) {
        filtered = filtered.filter(v => v.status === filters.statusFilter);
      }

      // Apply days filter
      if (filters.daysFilter) {
        filtered = filtered.filter(v => v.daysInStatus >= filters.daysFilter.min && v.daysInStatus <= filters.daysFilter.max);
      }

      // Apply search filter
      if (filters.searchFilter) {
        const searchLower = filters.searchFilter.toLowerCase();
        filtered = filtered.filter(v => v.unitNumber.toLowerCase().includes(searchLower));
      }

      // Render table rows
      const tbody = document.getElementById('vehicleTableBody');
      tbody.innerHTML = '';

      filtered.forEach(vehicle => {
        const row = document.createElement('tr');
        row.style.borderBottom = '1px solid var(--border)';
        row.innerHTML = `
          <td style="padding: 12px 16px; font: 500 14px var(--font-body); color: var(--fg);">${vehicle.unitNumber}</td>
          <td style="padding: 12px 16px; font: 14px var(--font-body); color: var(--fg);">${vehicle.make}</td>
          <td style="padding: 12px 16px; font: 14px var(--font-body); color: var(--fg);">${vehicle.type}</td>
          <td style="padding: 12px 16px; font: 14px var(--font-body); color: var(--fg);">${vehicle.inServiceDate}</td>
          <td style="padding: 12px 16px;"><span class="status-badge ${getStatusClass(vehicle.status)}">${vehicle.status}</span></td>
          <td style="padding: 12px 16px; text-align: center; font: 14px var(--font-body); color: var(--fg);">${vehicle.daysInStatus}</td>
          <td style="padding: 12px 16px; font: 14px var(--font-body); color: var(--muted);">${vehicle.mileage.toLocaleString()} mi</td>
          <td style="padding: 12px 16px; font: 14px var(--font-body); color: var(--fg);">${vehicle.lastService}</td>
          <td style="padding: 12px 16px; text-align: center;"><button class="btn btn-link" onclick="navigateTo('vehicle', '${vehicle.unitNumber}')" style="padding: 0; font-size: 12px;">View</button></td>
        `;
        row.onmouseover = function() { this.style.background = 'var(--surface)'; };
        row.onmouseout = function() { this.style.background = ''; };
        tbody.appendChild(row);
      });

      // Update filter chip active states
      document.querySelectorAll('[data-filter="type"]').forEach(chip => {
        const value = chip.getAttribute('data-value');
        if (filters.typeFilters.includes(value)) {
          chip.classList.add('active');
          chip.style.background = 'var(--accent)';
          chip.style.color = 'white';
          chip.style.borderColor = 'var(--accent)';
        } else {
          chip.classList.remove('active');
          chip.style.background = 'transparent';
          chip.style.color = 'var(--fg)';
          chip.style.borderColor = 'var(--border)';
        }
      });
    }

    // Toggle type filter
    function toggleTypeFilter(type) {
      const filters = appState.uiState.filters;
      const index = filters.typeFilters.indexOf(type);
      if (index > -1) {

    // Export vehicles to CSV
    function exportVehicles() {
      const filters = appState.uiState.filters;
      let filtered = appState.vehicles;

      if (filters.typeFilters.length > 0) {
        filtered = filtered.filter(v => filters.typeFilters.includes(v.type));
      }
      if (filters.statusFilter) {
        filtered = filtered.filter(v => v.status === filters.statusFilter);
      }
      if (filters.daysFilter) {
        filtered = filtered.filter(v => v.daysInStatus >= filters.daysFilter.min && v.daysInStatus <= filters.daysFilter.max);
      }
      if (filters.searchFilter) {
        const searchLower = filters.searchFilter.toLowerCase();
        filtered = filtered.filter(v => v.unitNumber.toLowerCase().includes(searchLower));
      }

      let csv = 'Unit Number,Make/Model,Type,In-Service Date,Status,Days in Status,Mileage,Last Service,VIN\n';
      filtered.forEach(v => {
        csv += `"${v.unitNumber}","${v.make}","${v.type}","${v.inServiceDate}","${v.status}",${v.daysInStatus},${v.mileage},"${v.lastService}","${v.vin}"\n`;
      });

      const blob = new Blob([csv], { type: 'text/csv' });
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `vehicles-export-${new Date().toISOString().split('T')[0]}.csv`;
      document.body.appendChild(a);
      a.click();
      window.URL.revokeObjectURL(url);
      document.body.removeChild(a);
      showToast('Vehicles exported successfully', 'success');
    }

    // Render inspections table
    function renderInspectionsTable() {
      const tbody = document.getElementById('inspectionsTableBody');
      if (!tbody) return;
      tbody.innerHTML = '';

      let filtered = appState.inspections;

      // Apply status filter
      const statusFilter = appState.uiState.inspectionFilters.statusFilter;
      if (statusFilter !== 'All') {
        filtered = filtered.filter(i => i.status === statusFilter);
      }

      // Apply type filter
      const typeFilter = appState.uiState.inspectionFilters.typeFilter;
      if (typeFilter !== 'All') {
        filtered = filtered.filter(i => i.type === typeFilter);
      }

      // Apply date filter (basic implementation)
      const dateFilter = appState.uiState.inspectionFilters.dateRangeFilter;
      const today = new Date('2025-01-13');
      if (dateFilter === 'thisWeek') {
        const weekStart = new Date(today);
        weekStart.setDate(weekStart.getDate() - weekStart.getDay());
        const weekEnd = new Date(weekStart);
        weekEnd.setDate(weekEnd.getDate() + 6);
        filtered = filtered.filter(i => {
          const d = new Date(i.scheduledDate);
          return d >= weekStart && d <= weekEnd;
        });
      } else if (dateFilter === 'thisMonth') {
        filtered = filtered.filter(i => {
          const d = new Date(i.scheduledDate);
          return d.getMonth() === today.getMonth() && d.getFullYear() === today.getFullYear();
        });
      }

      filtered.forEach(inspection => {
        const vehicle = appState.vehicles.find(v => v.unitNumber === inspection.vehicleId);
        if (!vehicle) return;

        const row = document.createElement('tr');
        row.style.borderBottom = '1px solid var(--border)';
        row.style.height = '32px';

        const statusColor = {
          'Scheduled': 'var(--muted)',
          'In Progress': 'var(--status-in-shop)',
          'Completed': 'var(--status-complete)'
        }[inspection.status] || 'var(--muted)';

        const findings = inspection.findings.length > 0 ? inspection.findings.join(', ') : '—';
        const actions = inspection.status === 'Scheduled' 
          ? `<button class="btn btn-link" onclick="openAssignInspectorModal('${inspection.id}', '${inspection.vehicleId}')" style="font-size: 12px; margin-right: 8px; padding: 0;">Assign</button><button class="btn btn-link" onclick="openInspectionDetailsModal('${inspection.id}')" style="font-size: 12px; padding: 0;">View</button>`
          : inspection.status === 'In Progress'
          ? `<button class="btn btn-link" onclick="markInspectionComplete('${inspection.id}')" style="font-size: 12px; margin-right: 8px; padding: 0;">Mark Complete</button><button class="btn btn-link" onclick="openInspectionDetailsModal('${inspection.id}')" style="font-size: 12px; padding: 0;">View</button>`
          : `<button class="btn btn-link" onclick="openInspectionDetailsModal('${inspection.id}')" style="font-size: 12px; padding: 0;">View</button>`;

        row.innerHTML = `
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${inspection.vehicleId}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${vehicle.type}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${inspection.type}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${inspection.scheduledDate}</td>
          <td style="padding: 8px 16px;">
            <span class="status-badge" style="background: ${statusColor}; color: white; padding: 4px 8px; border-radius: 4px; font: 12px var(--font-body); display: inline-block;">
              ${inspection.status}
            </span>
          </td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: ${inspection.inspector ? 'var(--fg)' : 'var(--muted)'};">

    let currentPartsSortColumn = null;
    let currentPartsSortDirection = 'asc';

    // Render parts table
    function renderPartsTable() {
      const tbody = document.getElementById('partsTableBody');
      if (!tbody) return;

      tbody.innerHTML = '';
      
      let sortedParts = [...partsData];
      if (currentPartsSortColumn) {
        sortedParts.sort((a, b) => {
          let aVal = a[currentPartsSortColumn];
          let bVal = b[currentPartsSortColumn];
          
          if (currentPartsSortColumn === 'unitCost' || currentPartsSortColumn === 'totalCost') {
            aVal = parseFloat(aVal) || 0;
            bVal = parseFloat(bVal) || 0;
          } else if (currentPartsSortColumn === 'qty') {
            aVal = parseInt(aVal) || 0;
            bVal = parseInt(bVal) || 0;
          }
          
          if (currentPartsSortDirection === 'asc') {
            return aVal > bVal ? 1 : aVal < bVal ? -1 : 0;
          } else {
            return aVal < bVal ? 1 : aVal > bVal ? -1 : 0;
          }
        });
      }

      sortedParts.forEach(part => {
        const totalCost = part.unitCost * part.qty;
        const row = document.createElement('tr');
        row.style.borderBottom = '1px solid var(--border)';
        row.style.height = '32px';
        
        row.innerHTML = `
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${part.partName}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${part.category}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg);">${part.vendor}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg); text-align: right;">$${part.unitCost.toFixed(2)}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg); text-align: center;">${part.qty}</td>
          <td style="padding: 8px 16px; font: 14px var(--font-body); color: var(--fg); text-align: right; font-weight: 600;">$${totalCost.toFixed(2)}</td>
          <td style="padding: 8px 16px; font: 12px var(--font-body); color: var(--muted); max-width: 150px; overflow: hidden; text-overflow: ellipsis;">${part.workOrders.join(', ')}</td>
          <td style="padding: 8px 16px; text-align: center;">
            <button onclick="navigateToWorkOrdersForPart('${part.workOrders.join(',')}')" style="background: transparent; border: none; color: var(--accent); cursor: pointer; font: 12px var(--font-body); text-decoration: underline;">View</button>
          </td>
        `;
        tbody.appendChild(row);
      });
    }

    // Sort table by column
    function sortTableByColumn(columnName) {
      if (currentPartsSortColumn === columnName) {
        currentPartsSortDirection = currentPartsSortDirection === 'asc' ? 'desc' : 'asc';
      } else {
        currentPartsSortColumn = columnName;
        currentPartsSortDirection = 'asc';
      }
      renderPartsTable();
    }

    // Export parts to CSV
    function exportCSV() {
      let csv = 'Part Name,Category,Vendor,Unit Cost,Qty,Total Cost,Work Orders\n';
      
      partsData.forEach(part => {
        const totalCost = part.unitCost * part.qty;
        const row = [

          part.qty,

      });

      const blob = new Blob([csv], { type: 'text/csv' });

      const a = document.createElement('a');
      a.href = url;

      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      window.URL.revokeObjectURL(url);
      
      showToast('Parts data exported to CSV', 'success');
